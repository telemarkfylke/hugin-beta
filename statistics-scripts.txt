
//  Buckets for antall medlinger i conversations
[
{ $group: { _id: "$conversationId", count: { $sum: 1 } } },
  { $bucket: {
      groupBy: "$count",
      boundaries: [1, 2, 3, 4, 5, 7, 10, 15, 20, 25, 30,  40, 50, 70, 100, 999999],
      output: { conversations: { $sum: 1 } }
  }},
  { $project: {
      _id: 0,
      //conversations: "$_id",
    lower: "$_id",
      upper: {
        $let: {
          vars: { b: [1, 2, 3, 4, 5, 7, 10, 15, 20, 25, 30,  40, 50, 70, 100, 999999] },
          in: {
            $arrayElemAt: [
              "$$b",
              { $add: [ { $indexOfArray: ["$$b", "$_id"] }, 1 ] }
            ]
          }
        }
      },
      conversations: 1
  }},
  { $group: {
      _id: null,
      items: { $push: { k: { $concat: [ { $toString: "$lower" }, "->", { $toString: "$upper" } ] }, v: "$conversations" } }
  }},
  { $replaceRoot: { newRoot: { $arrayToObject: "$items" } } }
]

      meldinger: { $sum: 1 }
  }},
  { $sort: { _id: 1 } },
  { $group: { _id: null, items: { $push: { k: { $toString: "$_id" }, v: "$meldinger" } } } },
  { $replaceRoot: { newRoot: { $arrayToObject: "$items" } } }
  
]


// Helg og ellers:
[
  { $match: { $expr: { $gte: [
    { $toDate: "$_id" },
    { $dateSubtract: { startDate: "$$NOW", unit: "day", amount: 7 } }
] } } },
  { $project: { t: { $toDate: "$_id" } } },
  { $project: {
      dag:  { $isoDayOfWeek: { date: "$t", timezone: "Europe/Oslo" } },   // 1 = mandag … 7 = søndag
      time: { $hour:         { date: "$t", timezone: "Europe/Oslo" } }
  }},
  { $group: {
      _id: { $switch: {
        branches: [
          { case: { $gte: ["$dag", 6] }, then: "helg" },
          { case: { $and: [ { $gte: ["$time", 7] }, { $lt: ["$time", 17] } ] }, then: "hverdag 07-17" },
          { case: { $and: [ { $gte: ["$time", 17] }, { $lte: ["$time", 23] } ] }, then: "kveld 17-23" }
        ],
        default: "hverdag natt"
      }},
      meldinger: { $sum: 1 }
  }},
  { $group: { _id: null, items: { $push: { k: "$_id", v: "$meldinger" } } } },
  { $replaceRoot: { newRoot: { $arrayToObject: "$items" } } }
]
