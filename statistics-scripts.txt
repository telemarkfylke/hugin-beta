
//  Buckets for antall medlinger i conversations
[
{ $group: { _id: "$conversationId", count: { $sum: 1 } } },
  { $bucket: {
      groupBy: "$count",
      boundaries: [1, 2, 3, 4, 5, 7, 10, 15, 20, 25, 30,  40, 50, 70, 100, 999999],
      output: { conversations: { $sum: 1 } }
  }},
  { $project: {
      _id: 0,
      //conversations: "$_id",
    lower: "$_id",
      upper: {
        $let: {
          vars: { b: [1, 2, 3, 4, 5, 7, 10, 15, 20, 25, 30,  40, 50, 70, 100, 999999] },
          in: {
            $arrayElemAt: [
              "$$b",
              { $add: [ { $indexOfArray: ["$$b", "$_id"] }, 1 ] }
            ]
          }
        }
      },
      conversations: 1
  }},
  { $group: {
      _id: null,
      items: { $push: { k: { $concat: [ { $toString: "$lower" }, "->", { $toString: "$upper" } ] }, v: "$conversations" } }
  }},
  { $replaceRoot: { newRoot: { $arrayToObject: "$items" } } }
]
